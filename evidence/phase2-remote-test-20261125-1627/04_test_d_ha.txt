events {
    worker_connections 1024;
}

http {
    upstream protocolx_backends {
        # Round-robin load balancing with passive health monitoring
        server 10.7.6.165:3001 max_fails=2 fail_timeout=10s;
        server 10.7.18.164:3002 max_fails=2 fail_timeout=10s;
    }

    server {
        listen 80;
        server_name app.protocolx.test;

        # Plain HTTP redirected automatically to HTTPS
        return 301 https://$host$request_uri;
    }

    server {
        listen 443 ssl;
        server_name app.protocolx.test;

        # TLS 1.2 / 1.3 Termination (Shagun's Mac 2 absolute path)
        ssl_certificate     /Users/Apple/Downloads/cn/tls/app.protocolx.test.crt;
        ssl_certificate_key /Users/Apple/Downloads/cn/tls/app.protocolx.test.key;
        ssl_protocols       TLSv1.2 TLSv1.3;
        ssl_ciphers         HIGH:!aNULL:!MD5;

        # Passive failover and upstream timeout tuning
        proxy_next_upstream error timeout http_502 http_503;
        proxy_connect_timeout 2s;
        proxy_read_timeout 5s;
        proxy_send_timeout 5s;

        # Standard reverse proxy headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;

        location /api/status {
            proxy_method GET;
            proxy_pass http://protocolx_backends;
            add_header Cache-Control "public, max-age=10";
            add_header X-Gateway "ProtocolX-Mac2-Edge";
        }

        location /api/health {
            proxy_pass http://protocolx_backends;
            add_header X-Gateway "ProtocolX-Mac2-Edge";
        }

        location / {
            proxy_pass http://protocolx_backends;
            add_header X-Gateway "ProtocolX-Mac2-Edge";
        }
    }
}
